⚙️ Connect Google Sheets

Follow these steps to send reports to your spreadsheet

1
Open your Google Sheet Click Extensions → Apps Script to open the script editor.
2
Paste this full script, then save (Ctrl+S) function doPost(e) { try { var sheet = SpreadsheetApp .openById("1j_gFds4AZ-NrbQkdkENO-u9I7psIqNutUx2trXEtf6g") .getActiveSheet(); // Body arrives as plain text in e.postData.contents var raw = (e && e.postData && e.postData.contents) ? e.postData.contents : null; if (!raw) throw new Error("No data received. Check deployment settings."); var d = JSON.parse(raw); if (sheet.getLastRow() === 0) { sheet.appendRow(["Timestamp","Date","School", "Location","Session Time","Facilitator", "Email","Phone","Zone","Male","Female", "Staff","Total","Topics","Challenges", "Highlights","Photo Count"]); } sheet.appendRow([ new Date().toISOString(), d.date||"", d.school||"", d.location||"", d.time||"", d.facilitator||"", d.email||"", d.phone||"", d.zone||"", d.male||0, d.female||0, d.staff||0, d.total||0, d.topics||"", d.challenges||"", d.highlights||"", d.photoCount||0 ]); return ContentService .createTextOutput(JSON.stringify({result:"success"})) .setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService .createTextOutput(JSON.stringify( {result:"error",message:err.toString()})) .setMimeType(ContentService.MimeType.JSON); } } function doOptions(e) { return ContentService .createTextOutput("") .setMimeType(ContentService.MimeType.TEXT); } function doGet() { return ContentService.createTextOutput("KCMC OK"); }
3
Deploy as Web App Click Deploy → New deployment → Web app.
Set "Execute as" = Me and "Who has access" = Anyone.
Click Deploy, authorize, then copy the deployment URL.

⚠️ Important: Every time you edit the script, create a New deployment — do not update the existing one. Apps Script caches the old version otherwise.
4
Paste your Web App URL below and click Save & Test
✅ Report submitted successfully!
