⚙️ Connect Google Sheets + Drive

Reports → Sheets · Photos → Google Drive with real links

✅ v11 — URL is pre-configured. The GAS endpoint is hardcoded in this file — no manual setup needed on any device. Use the fields below only if you need to change the URL or re-test.
1
Open your Google Sheet → Extensions → Apps Script → delete all code.
2
Paste the script and save (Ctrl+S) // ══════════════════════════════════════════════════════════════ // KCMC Discipleship — Google Apps Script v11 // Scopes declared explicitly in appsscript.json manifest. // IMPORTANT: Add these scopes to appsscript.json before deploying: // { // "oauthScopes": [ // "https://www.googleapis.com/auth/spreadsheets", // "https://www.googleapis.com/auth/drive", // "https://www.googleapis.com/auth/script.external_request" // ] // } // ══════════════════════════════════════════════════════════════ var SHEET_ID = "1j_gFds4AZ-NrbQkdkENO-u9I7psIqNutUx2trXEtf6g"; var FOLDER_NAME = "KCMCSessionPhotos"; function doAuthCheck() { var driveTest = DriveApp.getRootFolder().getName(); var sheetTest = SpreadsheetApp.openById(SHEET_ID).getName(); Logger.log("Auth OK — Drive: " + driveTest + " | Sheet: " + sheetTest); return "OK"; } function corsOutput(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.TEXT); } function doGet(e) { if (e && e.parameter && e.parameter.action === "ping") { return corsOutput({ result: "success", message: "KCMC OK v11" }); } return ContentService.createTextOutput("KCMC Discipleship GAS v11 — OK"); } function getOrCreateFolder(name) { var it = DriveApp.getFoldersByName(name); return it.hasNext() ? it.next() : DriveApp.createFolder(name); } function getOrCreateSubFolder(parent, name) { var it = parent.getFoldersByName(name); return it.hasNext() ? it.next() : parent.createFolder(name); } function doPost(e) { try { var raw = null; if (e.postData && e.postData.contents && e.postData.contents.length > 2) { raw = e.postData.contents; } if (!raw && e.parameter && e.parameter.data) { raw = e.parameter.data; } if (!raw) throw new Error("No data received"); var d = JSON.parse(raw); var action = d.action || "report"; if (action === "photo") { if (!d.data) throw new Error("Photo action missing base64 data"); var root = getOrCreateFolder(FOLDER_NAME); var folderName = (d.folderName || "Session").replace(/[\/\\:*?"<>|]/g, "-").slice(0, 100); var folder = getOrCreateSubFolder(root, folderName); var mime = d.mime || "image/jpeg"; var ext = mime.split("/")[1] || "jpg"; var bytes = Utilities.base64Decode(d.data); var blob = Utilities.newBlob(bytes, mime, "photo_" + (d.index || 1) + "." + ext); var file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); return corsOutput({ result: "success", url: file.getUrl(), name: file.getName() }); } var ss = SpreadsheetApp.openById(SHEET_ID); var sheet = ss.getActiveSheet(); if (sheet.getLastRow() === 0) { sheet.appendRow(["Timestamp","Date","School","Location","Session Time", "Facilitator","Email","Phone","Zone", "Male","Female","Staff","Total", "Topics","Challenges","Highlights","Photo Count","Photo Links (Drive)"]); sheet.setFrozenRows(1); sheet.getRange(1,1,1,18).setFontWeight("bold").setBackground("#1e4035").setFontColor("#ffffff"); } sheet.appendRow([ new Date().toISOString(), d.date||"", d.school||"", d.location||"", d.time||"", d.facilitator||"", d.email||"", d.phone||"", d.zone||"", Number(d.male)||0, Number(d.female)||0, Number(d.staff)||0, Number(d.total)||0, d.topics||"", d.challenges||"", d.highlights||"", Number(d.photoCount)||0, (d.photoLinks||[]).join("\n") ]); Logger.log("Report saved: " + d.school + " / " + d.facilitator); return corsOutput({ result: "success", saved: true }); } catch(err) { Logger.log("KCMC ERR: " + err.message + "\n" + err.stack); return ContentService .createTextOutput(JSON.stringify({ result: "error", message: err.toString() })) .setMimeType(ContentService.MimeType.TEXT); } }
3
Add OAuth scopes to appsscript.json
In the editor click ⚙️ Project Settings → check "Show appsscript.json". Then replace its contents with: { "timeZone": "Africa/Nairobi", "dependencies": {}, "exceptionLogging": "STACKDRIVER", "runtimeVersion": "V8", "oauthScopes": [ "https://www.googleapis.com/auth/spreadsheets", "https://www.googleapis.com/auth/drive", "https://www.googleapis.com/auth/script.external_request" ] }
4
Run doAuthCheck manually first
Select doAuthCheck in the dropdown → ▶ Run → Allow all permissions (Drive + Sheets must both appear). Check the log shows both Drive and Sheet names.
5
Deploy → Manage deployments → Edit → New version → Deploy
Execute as: Me  |  Who has access: Anyone
Always create a New version after any script change — old versions cache old scopes.
✅Done!
