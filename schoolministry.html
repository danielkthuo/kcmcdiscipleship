⚙️ Connect Google Sheets + Drive

Reports → Sheets · Photos → Google Drive

ℹ️ This is the v6 script — uses FormData POST so GAS reliably receives the body, then uploads each photo individually. You must re-deploy if you used an older version.
1
Open your Google Sheet → Extensions → Apps Script → delete all existing code.
2
Paste the script below and save (Ctrl+S) // ══════════════════════════════════════════════ // KCMC Discipleship — Apps Script v6 // FormData POST · per-photo uploads · Drive links in Sheet // ══════════════════════════════════════════════ var SHEET_ID = "YOUR_SHEET_ID_HERE"; // ← replace with your Sheet ID var FOLDER_NAME = "KCMC Session Photos"; function getOrCreateFolder(name) { var it = DriveApp.getFoldersByName(name); return it.hasNext() ? it.next() : DriveApp.createFolder(name); } function getOrCreateSubFolder(parent, name) { var it = parent.getFoldersByName(name); return it.hasNext() ? it.next() : parent.createFolder(name); } function doGet(e) { return ContentService.createTextOutput("KCMC OK v6"); } function doPost(e) { try { // Browser sends multipart/form-data → GAS exposes it as e.parameter.data // Fallback: text/plain body in e.postData.contents var raw = (e.parameter && e.parameter.data) ? e.parameter.data : (e.postData && e.postData.contents ? e.postData.contents : null); if (!raw) throw new Error("No data received"); var d = JSON.parse(raw); var action = d.action || "report"; // ── PHOTO: upload one image to Drive ─────────────── if (action === "photo") { var root = getOrCreateFolder(FOLDER_NAME); var folder = getOrCreateSubFolder(root, d.folderName || "Session"); var mime = d.mime || "image/jpeg"; var ext = mime.split("/")[1] || "jpg"; var bytes = Utilities.base64Decode(d.data); var blob = Utilities.newBlob(bytes, mime, "photo_" + d.index + "." + ext); var file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); return ok({ url: file.getUrl() }); } // ── REPORT: write one row to Sheets ──────────────── var sheet = SpreadsheetApp.openById(SHEET_ID).getActiveSheet(); if (sheet.getLastRow() === 0) { sheet.appendRow([ "Timestamp","Date","School","Location","Session Time", "Facilitator","Email","Phone","Zone", "Male","Female","Staff","Total", "Topics","Challenges","Highlights", "Photo Count","Photo Links (Drive)" ]); } sheet.appendRow([ new Date().toISOString(), d.date||"", d.school||"", d.location||"", d.time||"", d.facilitator||"", d.email||"", d.phone||"", d.zone||"", d.male||0, d.female||0, d.staff||0, d.total||0, d.topics||"", d.challenges||"", d.highlights||"", d.photoCount||0, (d.photoLinks||[]).join("\n") ]); return ok({ saved: true }); } catch(err) { Logger.log("ERR: " + err); return ContentService .createTextOutput(JSON.stringify({result:"error", message: err.toString()})) .setMimeType(ContentService.MimeType.JSON); } } function ok(obj) { obj.result = "success"; return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
3
Replace YOUR_SHEET_ID_HERE
From your Sheet URL: …/spreadsheets/d/SHEET_ID/edit
4
Deploy → New deployment → Web app
Execute as: Me  |  Who has access: Anyone
Authorize Drive + Sheets when prompted. Copy the /exec URL.
✅Done!
