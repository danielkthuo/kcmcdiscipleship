⚙️ Connect Google Sheets + Drive

Reports → Sheets · Photos → Google Drive

1
Open your Google Sheet Click Extensions → Apps Script to open the script editor. Delete any existing code.
2
Paste this script & save (Ctrl+S) // ── KCMC Discipleship Report Script ── var SHEET_ID = "1j_gFds4AZ-NrbQkdkENO-u9I7psIqNutUx2trXEtf6g"; var FOLDER_NAME = "KCMC Session Photos"; function getOrCreateFolder(name) { var it = DriveApp.getFoldersByName(name); return it.hasNext() ? it.next() : DriveApp.createFolder(name); } function doPost(e) { try { var sheet = SpreadsheetApp.openById(SHEET_ID).getActiveSheet(); // Accept both plain-body and form-encoded var raw = null; if (e.postData && e.postData.contents) raw = e.postData.contents; if (!raw && e.parameter && e.parameter.data) raw = e.parameter.data; if (!raw) throw new Error("No data received"); var d = JSON.parse(raw); // ── Save photos to Drive ── var photoLinks = []; if (d.photos && d.photos.length > 0) { var root = getOrCreateFolder(FOLDER_NAME); var subName = (d.school || "Session") + " – " + (d.date || "NoDate") + " – " + (d.facilitator || "Unknown"); var folder = root.createFolder(subName); d.photos.forEach(function(photo, i) { try { var mime = photo.mime || "image/jpeg"; var ext = mime.split("/")[1] || "jpg"; var bytes = Utilities.base64Decode(photo.data); var blob = Utilities.newBlob(bytes, mime, "photo_" + (i+1) + "." + ext); var file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); photoLinks.push(file.getUrl()); } catch(pe) { photoLinks.push("upload_error: " + pe.message); } }); } // ── Write to Sheet ── if (sheet.getLastRow() === 0) { sheet.appendRow([ "Timestamp","Date","School","Location","Session Time", "Facilitator","Email","Phone","Zone", "Male","Female","Staff","Total", "Topics","Challenges","Highlights","Photo Links" ]); } sheet.appendRow([ new Date().toISOString(), d.date||"", d.school||"", d.location||"", d.time||"", d.facilitator||"", d.email||"", d.phone||"", d.zone||"", d.male||0, d.female||0, d.staff||0, d.total||0, d.topics||"", d.challenges||"", d.highlights||"", photoLinks.join("\n") ]); return ContentService .createTextOutput(JSON.stringify({result:"success", photos: photoLinks.length})) .setMimeType(ContentService.MimeType.JSON); } catch(err) { Logger.log("ERROR: " + err.toString()); return ContentService .createTextOutput(JSON.stringify({result:"error", message: err.toString()})) .setMimeType(ContentService.MimeType.JSON); } } function doGet() { return ContentService.createTextOutput("KCMC OK"); }
3
Deploy as Web App Click Deploy → New deployment → Web app.
Set "Execute as" = Me & "Who has access" = Anyone.
Click Deploy — Google will ask to authorize Drive + Sheets access. Click Allow.
Copy the Web App URL.
